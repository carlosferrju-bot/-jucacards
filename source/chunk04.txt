    let d=state.decks.find(x=>x.id===editingDeck);
    const oldDisciplines=d.disciplines||[];

    d.desc=document.getElementById('deckDesc').value;
    d.color=document.getElementById('deckColor').value;
    d.disciplines=[...modalDisciplines];

    // They remain associated with the original discipline and can still be edited.

      toast('Concurso e disciplinas atualizados.');
    }

    state.decks.push({
      id:uid('d'),
      name,
      desc:document.getElementById('deckDesc').value,
      color:document.getElementById('deckColor').value,
      disciplines:[...modalDisciplines]
    });

let movingCardId=null;
let deletingDeckId=null;

function populateMoveDeckSelect(selectedId=''){
  const el=document.getElementById('moveCardDeck');
  if(!el)return;
  el.innerHTML=state.decks.map(d=>`<option value="${d.id}">${esc(d.name)}</option>`).join('');
  if(selectedId && [...el.options].some(o=>o.value===selectedId)) el.value=selectedId;
  populateMoveDisciplineSelect();
}

function populateMoveDisciplineSelect(selectedDiscipline=''){
  const deckId=document.getElementById('moveCardDeck')?.value;
  const el=document.getElementById('moveCardDiscipline');
  if(!el)return;
  const d=state.decks.find(x=>x.id===deckId);
  const disciplines=d?.disciplines||[];
  el.innerHTML=disciplines.length
    ? disciplines.map(x=>`<option value="${esc(x)}">${esc(x)}</option>`).join('')
    : '<option value="">Nenhuma disciplina cadastrada</option>';
  if(selectedDiscipline && disciplines.includes(selectedDiscipline)) el.value=selectedDiscipline;
}

function openMoveCardModal(id){
  const c=state.cards.find(x=>x.id===id);
  if(!c)return toast('Flashcard não encontrado.');
  movingCardId=id;

  const preview=document.getElementById('moveCardPreview');
  preview.innerHTML=`<b>${esc(strip(c.front)).slice(0,180)}</b><div class="muted" style="font-size:12px;margin-top:5px">${esc(deckName(c.deck))} · ${esc(c.discipline)}</div>`;

  populateMoveDeckSelect(c.deck);
  populateMoveDisciplineSelect(c.discipline);
  openModal('moveCardModal');
}

function saveMoveCard(){
  if(!movingCardId)return;
  const c=state.cards.find(x=>x.id===movingCardId);
  const deckId=document.getElementById('moveCardDeck')?.value;
  const discipline=document.getElementById('moveCardDiscipline')?.value;

  if(!c || !deckId)return toast('Selecione um concurso.');
  const d=state.decks.find(x=>x.id===deckId);
  if(!d)return toast('Concurso não encontrado.');
  if(!discipline)return toast('Selecione uma disciplina.');

  if(!Array.isArray(d.disciplines))d.disciplines=[];
  if(!d.disciplines.includes(discipline))d.disciplines.push(discipline);

  c.deck=deckId;
  c.discipline=discipline;

  closeModal('moveCardModal');
  movingCardId=null;
  persist();
  toast('Flashcard movido com sucesso.');
}

function openDeleteDeckModal(id){
  const d=state.decks.find(x=>x.id===id);
  if(!d)return toast('Concurso não encontrado.');

  deletingDeckId=id;
  const cards=state.cards.filter(c=>c.deck===id);
  const msg=document.getElementById('deleteDeckMessage');
  const wrap=document.getElementById('deleteDeckTargetWrap');
  const target=document.getElementById('deleteDeckTarget');

  if(cards.length){
    msg.textContent=`O concurso "${d.name}" possui ${cards.length} flashcard(s). Para excluí-lo sem perder esses cards, escolha outro concurso para receber todos eles.`;
    const others=state.decks.filter(x=>x.id!==id);
    target.innerHTML=others.length
      ? others.map(x=>`<option value="${x.id}">${esc(x.name)}</option>`).join('')
      : '';
    wrap.style.display=others.length?'grid':'none';
    document.getElementById('deleteDeckConfirm').textContent=others.length?'Excluir e transferir cards':'Excluir concurso';
  }else{
    msg.textContent=`O concurso "${d.name}" não possui flashcards. Ele poderá ser excluído sem transferência.`;
    wrap.style.display='none';
    target.innerHTML='';
    document.getElementById('deleteDeckConfirm').textContent='Excluir concurso';
  }

  openModal('deleteDeckModal');
}

function confirmDeleteDeck(){
  if(!deletingDeckId)return;
  const d=state.decks.find(x=>x.id===deletingDeckId);
  if(!d)return;

  const cards=state.cards.filter(c=>c.deck===deletingDeckId);
  if(cards.length){
    const targetId=document.getElementById('deleteDeckTarget')?.value;
    if(!targetId)return toast('Escolha o concurso que receberá os flashcards.');

    const target=state.decks.find(x=>x.id===targetId);
    if(!target)return toast('Concurso de destino não encontrado.');

    if(!Array.isArray(target.disciplines))target.disciplines=[];
    cards.forEach(c=>{
      c.deck=target.id;
      if(!target.disciplines.includes(c.discipline))target.disciplines.push(c.discipline);
    });
  }

  state.decks=state.decks.filter(x=>x.id!==deletingDeckId);
  closeModal('deleteDeckModal');
  deletingDeckId=null;
  persist();
  toast('Concurso excluído.');
}

function saveCard(){
  const deck=document.getElementById('cardDeck').value;
  const discipline=document.getElementById('cardDiscipline').value;
  const front=document.getElementById('cardFront').innerHTML.trim();
  const back=document.getElementById('cardBack').innerHTML.trim();
  if(!deck)return toast('Selecione um concurso.');
  if(!discipline)return toast('Selecione uma disciplina.');
  if(!strip(front).trim())return toast('Digite a pergunta.');
  if(!strip(back).trim())return toast('Digite a resposta.');

  const type=document.getElementById('cardType').value;
  const correct=document.getElementById('cardCorrect').value;
  const tags=document.getElementById('cardTags').value;
  const source=document.getElementById('cardSource').value;

  if(editingCard){
    let c=state.cards.find(x=>x.id===editingCard);
    c.deck=deck;c.discipline=discipline;c.front=front;c.back=back;c.type=type;c.correct=correct;c.tags=tags;c.source=source;
  }else{
    state.cards.push({id:uid('c'),deck,discipline,front,back,type,correct,tags,source,due:null,interval:0,reps:0,ease:2.5});
  }
  closeModal('cardModal');persist();toast(editingCard?'Flashcard atualizado.':'Flashcard criado.');
}
